#!/bin/bash
#==============================================================================
# Moodle 5.2 Site Provisioner
#
# Run this AFTER bootstrap.sh, once per Moodle instance you want to create.
# Each call spins up a fresh, isolated Moodle 5.2 site with its own:
#   - PostgreSQL database + user
#   - PHP-FPM pool (per-site resource isolation)
#   - Nginx server block + Let's Encrypt cert
#   - moodledata directory
#   - Redis session prefix (sessions auto-routed to Redis)
#   - Staggered cron job
#
# USAGE:
#   sudo ./provision-site.sh <slug> [admin-email]
#
#   <slug>          : 3-20 chars, lowercase letters/digits/hyphens.
#                     Used as subdomain, DB name, FPM pool, paths.
#   [admin-email]   : optional; defaults to LETSENCRYPT_EMAIL from bootstrap.
#
# EXAMPLE:
#   sudo ./provision-site.sh demo
#   → creates https://demo.ariise.cloud
#
# OUTPUT (saved to /etc/moodle-multi/sites/<slug>.conf):
#   - DB credentials
#   - Admin username + auto-generated password
#   - Site URL
#==============================================================================

set -euo pipefail

#==============================================================================
# Globals (loaded from bootstrap state)
#==============================================================================
STATE_DIR="/etc/moodle-multi"

if [[ ! -f "$STATE_DIR/global.conf" ]]; then
    echo "ERROR: bootstrap not complete. Run ./bootstrap.sh first." >&2
    exit 1
fi
# shellcheck disable=SC1091
source "$STATE_DIR/global.conf"

# Moodle branch — change here when 5.3 LTS is your target
MOODLE_BRANCH="MOODLE_502_STABLE"

WEB_USER="www-data"
LOG_FILE="/var/log/moodle-provision-$(date +%Y%m%d_%H%M%S).log"

RED='\033[0;31m'; GREEN='\033[0;32m'; YELLOW='\033[1;33m'; BLUE='\033[0;34m'; NC='\033[0m'
log()     { echo -e "$(date '+%F %T') [$1] $2" | tee -a "$LOG_FILE"; }
info()    { echo -e "${BLUE}[INFO]${NC}  $1"; log INFO "$1"; }
ok()      { echo -e "${GREEN}[ OK ]${NC}  $1"; log OK   "$1"; }
warn()    { echo -e "${YELLOW}[WARN]${NC}  $1"; log WARN "$1"; }
err()     { echo -e "${RED}[FAIL]${NC}  $1"; log FAIL "$1"; }
section() { echo; echo -e "${BLUE}==>${NC} $1"; log STEP "$1"; }

trap 'err "Provision failed at line $LINENO (exit $?). See $LOG_FILE"' ERR

#==============================================================================
# Argument parsing + validation
#==============================================================================
[[ $EUID -eq 0 ]] || { err "Must run as root"; exit 1; }

if [[ $# -lt 1 ]]; then
    cat <<EOF
Usage: $0 <slug> [admin-email]

  slug         3-20 chars, lowercase a-z 0-9 and hyphens (must start with a letter)
  admin-email  defaults to bootstrap email if omitted

Example: $0 demo
EOF
    exit 1
fi

SLUG="$1"
ADMIN_EMAIL="${2:-$LETSENCRYPT_EMAIL}"

# Validate slug
if [[ ! "$SLUG" =~ ^[a-z][a-z0-9-]{2,19}$ ]]; then
    err "Invalid slug '$SLUG': must be 3-20 chars, lowercase a-z/0-9/hyphens, start with a letter"
    exit 1
fi

# Db prefix (Moodle limit is 10 chars including trailing _, so prefix <= 9 chars)
DB_PREFIX="mdl_"

# Derived names
DOMAIN="${SLUG}.${BASE_DOMAIN}"
SITE_DIR="/var/www/moodle/${SLUG}"
DATA_DIR="/var/moodledata/${SLUG}"
DB_NAME="moodle_${SLUG}"
DB_USER="moodle_${SLUG}"
FPM_POOL="moodle_${SLUG}"
FPM_SOCK="/run/php/php${PHP_VERSION}-fpm-${SLUG}.sock"
NGINX_CONF="/etc/nginx/sites-available/moodle-${SLUG}"
SITE_CONF="$STATE_DIR/sites/${SLUG}.conf"
CRON_FILE="/etc/cron.d/moodle-${SLUG}"

# Refuse if site already exists
if [[ -f "$SITE_CONF" ]] || [[ -d "$SITE_DIR" ]] || [[ -d "$DATA_DIR" ]]; then
    err "Site '$SLUG' appears to already exist."
    err "  Site config : $SITE_CONF"
    err "  Code dir    : $SITE_DIR"
    err "  Data dir    : $DATA_DIR"
    err "Remove these first to reprovision (or use a different slug)."
    exit 1
fi

#==============================================================================
# Generate secrets
#==============================================================================
gen_password() { openssl rand -base64 24 | tr -d '/+=' | head -c 24; }

DB_PASS=$(gen_password)
ADMIN_USER="admin"
ADMIN_PASS=$(gen_password)

#==============================================================================
# 1. DNS sanity-check
#==============================================================================
step_dns_check() {
    section "STEP 1/10 — DNS check for $DOMAIN"
    local server_ip resolved_ip
    server_ip=$(curl -fsSL https://api.ipify.org 2>/dev/null || hostname -I | awk '{print $1}')
    resolved_ip=$(dig +short "$DOMAIN" | tail -n1)

    if [[ -z "$resolved_ip" ]]; then
        err "DNS for $DOMAIN does not resolve."
        err "Ensure wildcard *.${BASE_DOMAIN} A-record points to $server_ip"
        exit 1
    fi
    if [[ "$resolved_ip" != "$server_ip" ]]; then
        warn "DNS resolves $DOMAIN → $resolved_ip but server IP is $server_ip"
        warn "Let's Encrypt may fail if DNS hasn't fully propagated. Continuing..."
    else
        ok "DNS: $DOMAIN → $resolved_ip ✓"
    fi
}

#==============================================================================
# 2. PostgreSQL DB + user
#==============================================================================
step_database() {
    section "STEP 2/10 — PostgreSQL database + user"

    sudo -u postgres psql <<SQL
CREATE USER ${DB_USER} WITH PASSWORD '${DB_PASS}';
CREATE DATABASE ${DB_NAME}
    WITH OWNER = ${DB_USER}
    ENCODING = 'UTF8'
    LC_COLLATE = 'en_US.UTF-8'
    LC_CTYPE = 'en_US.UTF-8'
    TEMPLATE = template0;
GRANT ALL PRIVILEGES ON DATABASE ${DB_NAME} TO ${DB_USER};
\\c ${DB_NAME}
GRANT ALL ON SCHEMA public TO ${DB_USER};
SQL
    ok "Database '$DB_NAME' created with owner '$DB_USER'"
}

#==============================================================================
# 3. Clone Moodle 5.2 + composer install
#==============================================================================
step_clone_moodle() {
    section "STEP 3/10 — Clone Moodle ($MOODLE_BRANCH)"

    mkdir -p "$(dirname "$SITE_DIR")"
    git clone --depth 1 --branch "$MOODLE_BRANCH" \
        https://github.com/moodle/moodle.git "$SITE_DIR"
    ok "Moodle 5.2 cloned to $SITE_DIR"

    # Composer install (Moodle 5.2 vendors third-party libs via Composer)
    if [[ -f "$SITE_DIR/composer.json" ]]; then
        if ! command -v composer &>/dev/null; then
            info "Installing Composer..."
            curl -sS https://getcomposer.org/installer | php -- \
                --install-dir=/usr/local/bin --filename=composer
        fi
        info "Running composer install (may take a minute)..."
        cd "$SITE_DIR"
        sudo -u "$WEB_USER" HOME=/tmp composer install \
            --no-dev --classmap-authoritative --no-interaction --no-progress \
            2>&1 | tee -a "$LOG_FILE" || warn "composer install reported issues — continuing"
        cd - >/dev/null
        ok "Composer dependencies installed"
    fi
}

#==============================================================================
# 4. Permissions (with ACL — solves the plugin install problem)
#==============================================================================
step_permissions() {
    section "STEP 4/10 — Permissions + ACLs for plugin directories"

    # moodledata: writable by web user
    mkdir -p "$DATA_DIR"
    chown -R "$WEB_USER:$WEB_USER" "$DATA_DIR"
    chmod 0770 "$DATA_DIR"

    # Code: owned by root, readable by web user
    chown -R root:"$WEB_USER" "$SITE_DIR"
    find "$SITE_DIR" -type d -exec chmod 0755 {} \;
    find "$SITE_DIR" -type f -exec chmod 0644 {} \;

    # Plugin directories need write access from www-data so admin UI can install plugins
    # ACL with default propagation (-d) ensures NEW files inherit the permission
    local plugin_dirs=(
        public/mod public/local public/blocks public/theme
        public/auth public/filter public/admin/tool
        public/course/format public/question/type public/repository
        public/availability/condition public/customfield/field
        public/enrol public/grade/import public/grade/export
        public/lib/editor/tiny/plugins public/message/output
        public/portfolio public/cache/stores
        /admin/cli public/admin
    )
    for d in "${plugin_dirs[@]}"; do
        if [[ -d "$SITE_DIR/$d" ]]; then
            setfacl -R -m u:${WEB_USER}:rwX "$SITE_DIR/$d"
            setfacl -R -d -m u:${WEB_USER}:rwX "$SITE_DIR/$d"
        fi
    done
    ok "Permissions set; web user has write ACL on plugin directories"
}

#==============================================================================
# 5. Per-site PHP-FPM pool
#==============================================================================
step_fpm_pool() {
    section "STEP 5/10 — Dedicated PHP-FPM pool"

    # Sized for ~5 sites on $TOTAL_RAM_GB GB box.
    # Per-site PHP cap ≈ (RAM_GB - 4) / 5 GB ⇒ ~14 children at ~80MB each
    cat > "/etc/php/${PHP_VERSION}/fpm/pool.d/${FPM_POOL}.conf" <<EOF
[${FPM_POOL}]
user = ${WEB_USER}
group = ${WEB_USER}
listen = ${FPM_SOCK}
listen.owner = ${WEB_USER}
listen.group = ${WEB_USER}
listen.mode = 0660

pm = dynamic
pm.max_children = 14
pm.start_servers = 4
pm.min_spare_servers = 2
pm.max_spare_servers = 6
pm.max_requests = 500
pm.process_idle_timeout = 30s

request_terminate_timeout = 300s
request_slowlog_timeout = 10s
slowlog = /var/log/php-fpm/${SLUG}-slow.log

php_admin_value[memory_limit] = 384M
php_admin_value[max_input_vars] = 5000
php_admin_value[upload_max_filesize] = 256M
php_admin_value[post_max_size] = 256M
php_admin_value[max_execution_time] = 300
php_admin_value[error_log] = /var/log/php-fpm/${SLUG}-error.log
php_admin_flag[log_errors] = on
php_admin_value[date.timezone] = Asia/Kolkata
security.limit_extensions = .php
EOF

    mkdir -p /var/log/php-fpm
    chown "${WEB_USER}:${WEB_USER}" /var/log/php-fpm

    systemctl reload "php${PHP_VERSION}-fpm"
    sleep 1
    [[ -S "$FPM_SOCK" ]] || { err "FPM socket $FPM_SOCK not created"; exit 1; }
    ok "PHP-FPM pool '$FPM_POOL' running on $FPM_SOCK"
}

#==============================================================================
# 6. Nginx vhost (HTTP first — Certbot will inject HTTPS block)
#==============================================================================
step_nginx_vhost() {
    section "STEP 6/10 — Nginx server block (HTTP)"

    cat > "$NGINX_CONF" <<EOF
# Moodle 5.2 site: ${SLUG}
# Document root → /public (Moodle 5.1+ requirement)
server {
    listen 80;
    listen [::]:80;
    server_name ${DOMAIN};

    root ${SITE_DIR}/public;
    index index.php;

    access_log /var/log/nginx/${SLUG}_access.log;
    error_log  /var/log/nginx/${SLUG}_error.log;

    # Moodle 5.2 routing engine — try real file, then fall through to /r.php
    location / {
        try_files \$uri \$uri/ /r.php\$is_args\$args;
    }

    # PHP handler — supports slash-args used by Moodle router
    location ~ [^/]\.php(/|\$) {
        fastcgi_split_path_info ^(.+?\.php)(/.*)\$;
        set \$path_info \$fastcgi_path_info;
        try_files \$fastcgi_script_name \$fastcgi_script_name/ /r.php\$is_args\$args;

        fastcgi_pass unix:${FPM_SOCK};
        include fastcgi_params;
        fastcgi_param PATH_INFO \$path_info;
        fastcgi_param SCRIPT_FILENAME \$realpath_root\$fastcgi_script_name;
        fastcgi_param DOCUMENT_ROOT \$realpath_root;
        fastcgi_read_timeout 300;
        fastcgi_buffers 16 16k;
        fastcgi_buffer_size 32k;
    }

    # Login bruteforce rate-limit
    location ~ ^/(login/index\.php|login/token\.php) {
        limit_req zone=moodle_login burst=10 nodelay;
        try_files \$uri \$uri/ /r.php\$is_args\$args;
    }

    # XSendfile — efficient file delivery via Nginx X-Accel-Redirect
    location /dataroot/ {
        internal;
        alias ${DATA_DIR}/;
    }

    # Static asset caching
    location ~* \\.(jpg|jpeg|png|gif|ico|css|js|woff2?|svg)\$ {
        expires 30d;
        add_header Cache-Control "public, immutable";
        try_files \$uri =404;
    }

    # Hide dotfiles, allow .well-known (RFC 5785) for ACME challenges
    location ~ /\\.(?!well-known).* {
        return 404;
    }

    # Block sensitive Moodle internals (per official Moodle Nginx docs)
    location ~ (/vendor/|/node_modules/|composer\\.json|/readme|/README|readme\\.txt|/upgrade\\.txt|/UPGRADING\\.md|db/install\\.xml|/fixtures/|/behat/|phpunit\\.xml|\\.lock|environment\\.xml) {
        deny all;
        return 404;
    }

    # Security headers
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;
}
EOF

    ln -sf "$NGINX_CONF" "/etc/nginx/sites-enabled/moodle-${SLUG}"
    nginx -t
    systemctl reload nginx
    ok "Nginx vhost active for http://${DOMAIN}"
}

#==============================================================================
# 7. Let's Encrypt (HTTP-01 challenge)
#==============================================================================
step_certbot() {
    section "STEP 7/10 — Let's Encrypt SSL"

    certbot --nginx \
        --non-interactive \
        --agree-tos \
        --redirect \
        --email "$ADMIN_EMAIL" \
        -d "$DOMAIN" \
        2>&1 | tee -a "$LOG_FILE"
    ok "HTTPS active for https://${DOMAIN}"
}

#==============================================================================
# 8. Moodle CLI install
#==============================================================================
step_moodle_install() {
    section "STEP 8/10 — Moodle CLI install"

    # Brief permissions loosen for install — installer writes config.php
    chown "$WEB_USER" "$SITE_DIR"

    sudo -u "$WEB_USER" /usr/bin/php${PHP_VERSION} "$SITE_DIR/admin/cli/install.php" \
        --non-interactive \
        --agree-license \
        --lang=en \
        --wwwroot="https://${DOMAIN}" \
        --dataroot="$DATA_DIR" \
        --dbtype=pgsql \
        --dbhost=localhost \
        --dbname="$DB_NAME" \
        --dbuser="$DB_USER" \
        --dbpass="$DB_PASS" \
        --prefix="$DB_PREFIX" \
        --fullname="${SLUG^} Moodle" \
        --shortname="${SLUG^}" \
        --summary="Moodle 5.2 instance for ${SLUG}" \
        --adminuser="$ADMIN_USER" \
        --adminpass="$ADMIN_PASS" \
        --adminemail="$ADMIN_EMAIL" \
        2>&1 | tee -a "$LOG_FILE"

    chown root "$SITE_DIR"
    [[ -f "$SITE_DIR/config.php" ]] || { err "config.php was not created — install failed"; exit 1; }
    ok "Moodle installed; admin user '$ADMIN_USER' created"
}

#==============================================================================
# 9. Patch config.php — routing, Redis sessions, XSendfile, SSL proxy
#==============================================================================
step_patch_config() {
    section "STEP 9/10 — Patch config.php (router, Redis, XSendfile)"

    local cfg="$SITE_DIR/config.php"
    local marker="// === moodle-multi additions ==="
    if grep -q "$marker" "$cfg"; then
        warn "config.php already patched — skipping"
        return
    fi

    # Insert additions before the require_once line (Moodle puts that last)
    local tmp=$(mktemp)
    awk -v add="$(cat <<EOF

$marker
// Moodle 5.2 routing engine — declare router as configured
\$CFG->routerconfigured = true;

// XSendfile via Nginx X-Accel-Redirect (faster file serving)
\$CFG->xsendfile = 'X-Accel-Redirect';
\$CFG->xsendfilealiases = array(
    '/dataroot/' => \$CFG->dataroot,
);

// Redis-backed sessions (avoids on-disk session contention across PHP-FPM workers)
\$CFG->session_handler_class = '\\\\core\\\\session\\\\redis';
\$CFG->session_redis_host = '127.0.0.1';
\$CFG->session_redis_port = 6379;
\$CFG->session_redis_database = 0;
\$CFG->session_redis_prefix = 'mdl_${SLUG}_sess_';
\$CFG->session_redis_acquire_lock_timeout = 120;
\$CFG->session_redis_lock_expire = 7200;

// Reverse-proxy / SSL handling (in case you front this with a proxy/CDN later)
\$CFG->sslproxy = false;
EOF
)" '
        /^require_once/ && !done { print add; done=1 }
        { print }
    ' "$cfg" > "$tmp"
    mv "$tmp" "$cfg"
    chown root:"$WEB_USER" "$cfg"
    chmod 0640 "$cfg"

    # Purge caches so changes take effect
    sudo -u "$WEB_USER" /usr/bin/php${PHP_VERSION} "$SITE_DIR/admin/cli/purge_caches.php" \
        2>&1 | tee -a "$LOG_FILE" || true
    ok "config.php patched + caches purged"

    info "Note: configure Redis as MUC application cache in:"
    info "      Site administration → Plugins → Caching → Configure stores"
}

#==============================================================================
# 10. Cron with stagger
#==============================================================================
step_cron() {
    section "STEP 10/10 — Cron job (staggered)"

    # Spread crons across the minute so 5 sites don't fire simultaneously
    local existing
    existing=$(find "$STATE_DIR/sites/" -maxdepth 1 -name '*.conf' -type f 2>/dev/null | wc -l)
    local offset=$(( (existing * 12) % 60 ))

    cat > "$CRON_FILE" <<EOF
# Moodle cron for ${SLUG} — staggered by ${offset}s
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
* * * * * ${WEB_USER} sleep ${offset}; /usr/bin/php${PHP_VERSION} ${SITE_DIR}/admin/cli/cron.php >/dev/null 2>&1
EOF
    chmod 0644 "$CRON_FILE"
    ok "Cron added (${offset}s offset) → $CRON_FILE"
}

#==============================================================================
# Save site config (credentials!)
#==============================================================================
save_site_conf() {
    cat > "$SITE_CONF" <<EOF
# Moodle site '${SLUG}' — generated $(date)
SLUG="${SLUG}"
DOMAIN="${DOMAIN}"
URL="https://${DOMAIN}"

SITE_DIR="${SITE_DIR}"
DATA_DIR="${DATA_DIR}"

DB_NAME="${DB_NAME}"
DB_USER="${DB_USER}"
DB_PASS="${DB_PASS}"

FPM_POOL="${FPM_POOL}"
FPM_SOCK="${FPM_SOCK}"

ADMIN_USER="${ADMIN_USER}"
ADMIN_PASS="${ADMIN_PASS}"
ADMIN_EMAIL="${ADMIN_EMAIL}"

NGINX_CONF="${NGINX_CONF}"
CRON_FILE="${CRON_FILE}"
EOF
    chmod 0600 "$SITE_CONF"
    ok "Site credentials saved → $SITE_CONF"
}

#==============================================================================
# Final summary
#==============================================================================
print_summary() {
cat <<EOF

================================================================================
$(echo -e "${GREEN}MOODLE SITE PROVISIONED — ${SLUG}${NC}")
================================================================================

  URL                : https://${DOMAIN}
  Admin user         : ${ADMIN_USER}
  Admin password     : ${ADMIN_PASS}
  Admin email        : ${ADMIN_EMAIL}

  Code               : ${SITE_DIR}
  Data               : ${DATA_DIR}
  Database           : ${DB_NAME} (user: ${DB_USER})
  PHP-FPM pool       : ${FPM_POOL}
  Cron               : ${CRON_FILE}

  Credentials saved  : ${SITE_CONF}
  Provision log      : ${LOG_FILE}

NEXT (one-time):
  1. Visit https://${DOMAIN} and log in to verify.
  2. Site administration → Plugins → Caching → Configure stores
     Add a Redis store, then map "Application" + "Request" cache definitions to it.
  3. Site administration → Server → System paths → set /usr/bin/du, /usr/bin/aspell, etc.

To provision another site:
     sudo $0 <other-slug>

================================================================================
EOF
}

#==============================================================================
# Main
#==============================================================================
main() {
    info "Provisioning Moodle 5.2 site: ${SLUG} → https://${DOMAIN}"
    info "Logging to $LOG_FILE"

    step_dns_check
    step_database
    step_clone_moodle
    step_permissions
    step_fpm_pool
    step_nginx_vhost
    step_certbot
    step_moodle_install
    step_patch_config
    step_cron

    save_site_conf
    print_summary
}

main "$@"
