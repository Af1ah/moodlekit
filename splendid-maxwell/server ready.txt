#!/bin/bash
#==============================================================================
# Moodle 5.2 Multi-Tenant Bootstrap Script
#
# Run ONCE on a fresh Ubuntu 22.04 / 24.04 server.
# Installs and configures all shared services for hosting multiple Moodle 5.2
# sites on one box (PHP 8.4 + Nginx + PostgreSQL 17 + Redis + Certbot).
#
# After this completes, use ./provision-site.sh <slug> to create individual
# Moodle instances.
#
# Tested target: 16 GB RAM VPS, ~5 sites, ~300 active users total.
#
# Author: built for Aflah / Arrise — 2026
#==============================================================================

set -euo pipefail

#==============================================================================
# CONFIGURATION — edit these before first run
#==============================================================================

# Base domain. Subdomains like <slug>.ariise.cloud will be created automatically
# REQUIREMENT: set up wildcard DNS *.ariise.cloud → this server's public IP
BASE_DOMAIN="ariise.cloud"

# Email used for Let's Encrypt registration & expiry warnings
LETSENCRYPT_EMAIL="muhammadaflah23524@gmail.com"

# PHP version — Moodle 5.2 supports 8.3 / 8.4. DO NOT use 8.5 (not supported).
PHP_VERSION="8.4"

# PostgreSQL major version — Moodle 5.2 requires 16+
PG_VERSION="17"

# Total system RAM in GB. Used to size Postgres + Redis + PHP-FPM.
TOTAL_RAM_GB="16"

# State / shared config directory used by both scripts
STATE_DIR="/etc/moodle-multi"

#==============================================================================
# Internal — do not edit below this line unless you know what you're doing
#==============================================================================
LOG_FILE="/var/log/moodle-bootstrap-$(date +%Y%m%d_%H%M%S).log"
RED='\033[0;31m'; GREEN='\033[0;32m'; YELLOW='\033[1;33m'; BLUE='\033[0;34m'; NC='\033[0m'

log()     { echo -e "$(date '+%F %T') [$1] $2" | tee -a "$LOG_FILE"; }
info()    { echo -e "${BLUE}[INFO]${NC}  $1"; log INFO "$1"; }
ok()      { echo -e "${GREEN}[ OK ]${NC}  $1"; log OK   "$1"; }
warn()    { echo -e "${YELLOW}[WARN]${NC}  $1"; log WARN "$1"; }
err()     { echo -e "${RED}[FAIL]${NC}  $1"; log FAIL "$1"; }
section() { echo; echo -e "${BLUE}==>${NC} $1"; log STEP "$1"; }

trap 'err "Script failed at line $LINENO (exit $?). See $LOG_FILE"' ERR

require_root() {
    [[ $EUID -eq 0 ]] || { err "Must run as root (use sudo)"; exit 1; }
}

already_done() {
    # Idempotency marker — bootstrap only runs once
    if [[ -f "$STATE_DIR/.bootstrap_complete" ]]; then
        warn "Bootstrap was already completed on $(cat $STATE_DIR/.bootstrap_complete)"
        read -p "Re-run anyway? This may overwrite tuning configs. [y/N] " ans
        [[ "$ans" =~ ^[Yy]$ ]] || exit 0
    fi
}

#==============================================================================
# 1. System update + base utilities
#==============================================================================
step_system() {
    section "STEP 1/9 — System update + base packages"
    export DEBIAN_FRONTEND=noninteractive
    apt-get update -qq
    apt-get upgrade -y -qq
    apt-get install -y -qq \
        curl wget git unzip jq dnsutils \
        software-properties-common apt-transport-https ca-certificates gnupg lsb-release \
        ufw fail2ban cron acl logrotate \
        build-essential pkg-config
    ok "Base packages installed"
}

#==============================================================================
# 2. PHP 8.4 + extensions required by Moodle 5.2
#==============================================================================
step_php() {
    section "STEP 2/9 — PHP $PHP_VERSION + Moodle 5.2 required extensions"

    if ! grep -rq "ondrej/php" /etc/apt/sources.list.d/ 2>/dev/null; then
        add-apt-repository -y ppa:ondrej/php
        apt-get update -qq
    fi

    # Extensions required by Moodle 5.2:
    #   sodium (MANDATORY since 5.1)
    #   intl, mbstring, xml, curl, gd, zip, soap, opcache (still in 8.4)
    #   pgsql for PostgreSQL, mysql for MariaDB sites if needed
    #   exif, bcmath, readline (recommended)
    #   redis (for session + MUC cache)
    apt-get install -y -qq \
        php${PHP_VERSION}-fpm \
        php${PHP_VERSION}-cli \
        php${PHP_VERSION}-common \
        php${PHP_VERSION}-curl \
        php${PHP_VERSION}-gd \
        php${PHP_VERSION}-intl \
        php${PHP_VERSION}-mbstring \
        php${PHP_VERSION}-xml \
        php${PHP_VERSION}-xmlrpc \
        php${PHP_VERSION}-zip \
        php${PHP_VERSION}-soap \
        php${PHP_VERSION}-pgsql \
        php${PHP_VERSION}-mysql \
        php${PHP_VERSION}-bcmath \
        php${PHP_VERSION}-readline \
        php${PHP_VERSION}-opcache \
        php${PHP_VERSION}-redis \
        php${PHP_VERSION}-exif \
        php${PHP_VERSION}-ldap

    # Verify sodium (required by Moodle 5.2)
    if ! php${PHP_VERSION} -m | grep -qi '^sodium$'; then
        err "PHP sodium extension missing — Moodle 5.2 requires it"
        exit 1
    fi
    ok "PHP $PHP_VERSION installed with sodium + all Moodle extensions"

    # ---- Global PHP tuning (CLI + FPM) ----
    # Moodle 5.2 needs max_input_vars >= 5000
    for ini in /etc/php/${PHP_VERSION}/fpm/php.ini /etc/php/${PHP_VERSION}/cli/php.ini; do
        cp -n "$ini" "${ini}.orig" || true
        sed -i \
            -e 's|^;*\s*max_input_vars\s*=.*|max_input_vars = 5000|' \
            -e 's|^;*\s*memory_limit\s*=.*|memory_limit = 384M|' \
            -e 's|^;*\s*upload_max_filesize\s*=.*|upload_max_filesize = 256M|' \
            -e 's|^;*\s*post_max_size\s*=.*|post_max_size = 256M|' \
            -e 's|^;*\s*max_execution_time\s*=.*|max_execution_time = 300|' \
            -e 's|^;*\s*max_input_time\s*=.*|max_input_time = 600|' \
            -e 's|^;*\s*default_socket_timeout\s*=.*|default_socket_timeout = 300|' \
            -e 's|^;*\s*date.timezone\s*=.*|date.timezone = Asia/Kolkata|' \
            "$ini"
        # Append if missing
        grep -q "^max_input_vars" "$ini" || echo "max_input_vars = 5000" >> "$ini"
    done

    # ---- OPcache + JIT (huge perf win for Moodle) ----
    cat > /etc/php/${PHP_VERSION}/fpm/conf.d/10-opcache-moodle.ini <<'EOF'
; Moodle 5.2 — OPcache + JIT tuning
opcache.enable=1
opcache.enable_cli=0
opcache.memory_consumption=256
opcache.interned_strings_buffer=16
opcache.max_accelerated_files=20000
opcache.validate_timestamps=1
opcache.revalidate_freq=60
opcache.save_comments=1
opcache.fast_shutdown=1
opcache.jit_buffer_size=128M
opcache.jit=1255
EOF

    # ---- Default www pool: keep it alive but trim it down ----
    # PHP-FPM refuses to start with zero pools, so we leave the default www.conf
    # in place. Nothing in our nginx configs ever points to its socket — each
    # site uses its own dedicated socket — so this pool just idles harmlessly.
    # We do shrink it so it doesn't waste resources.
    if [[ -f /etc/php/${PHP_VERSION}/fpm/pool.d/www.conf ]]; then
        sed -i \
            -e 's|^pm =.*|pm = ondemand|' \
            -e 's|^pm.max_children =.*|pm.max_children = 2|' \
            -e 's|^;*pm.process_idle_timeout =.*|pm.process_idle_timeout = 10s|' \
            /etc/php/${PHP_VERSION}/fpm/pool.d/www.conf
        ok "Default PHP-FPM pool shrunk (ondemand, max 2 workers — idle/unused)"
    fi

    systemctl enable --now php${PHP_VERSION}-fpm
    systemctl restart php${PHP_VERSION}-fpm
    ok "PHP-FPM tuned and running"
}

#==============================================================================
# 3. PostgreSQL — tuned for the box size
#==============================================================================
step_postgres() {
    section "STEP 3/9 — PostgreSQL $PG_VERSION (Moodle 5.2 requires PG 16+)"

    if ! command -v psql &>/dev/null; then
        apt-get install -y -qq postgresql-common
        # Auto-add PGDG repo non-interactively
        yes "" | /usr/share/postgresql-common/pgdg/apt.postgresql.org.sh -y || true
        apt-get update -qq
        apt-get install -y -qq "postgresql-${PG_VERSION}" "postgresql-contrib-${PG_VERSION}"
    fi

    systemctl enable --now postgresql
    sleep 2

    # ---- Tuning (sized for $TOTAL_RAM_GB GB box, ~5 Moodle sites) ----
    local pg_conf_dir="/etc/postgresql/${PG_VERSION}/main/conf.d"
    mkdir -p "$pg_conf_dir"

    # Calculate sizes — leave RAM for PHP-FPM & OS
    local shared_buffers=$(( TOTAL_RAM_GB * 1024 / 5 ))     # ~20% RAM
    local effective_cache=$(( TOTAL_RAM_GB * 1024 / 2 ))    # ~50% RAM
    cat > "$pg_conf_dir/moodle-tuning.conf" <<EOF
# Moodle multi-tenant tuning — sized for ${TOTAL_RAM_GB}GB RAM
listen_addresses = 'localhost'
shared_buffers = ${shared_buffers}MB
effective_cache_size = ${effective_cache}MB
work_mem = 16MB
maintenance_work_mem = 512MB
max_connections = 200
random_page_cost = 1.1
effective_io_concurrency = 200
wal_buffers = 16MB
checkpoint_completion_target = 0.9
default_statistics_target = 100
max_wal_size = 2GB
min_wal_size = 512MB
log_min_duration_statement = 1000
log_line_prefix = '%t [%p] %q%u@%d '
EOF

    systemctl restart postgresql
    sleep 2
    sudo -u postgres psql -c "SELECT version();" >/dev/null
    ok "PostgreSQL $PG_VERSION installed and tuned (shared_buffers=${shared_buffers}MB)"

    # Save the postgres superuser is local-only (peer auth) — provision script uses sudo -u postgres
    info "Postgres uses peer authentication for the postgres user; provision script handles per-site DBs"
}

#==============================================================================
# 4. Nginx — base install + per-site rate-limit zone
#==============================================================================
step_nginx() {
    section "STEP 4/9 — Nginx + global config"

    apt-get install -y -qq nginx

    # ---- Global tuning snippet ----
    cat > /etc/nginx/conf.d/10-moodle-global.conf <<'EOF'
# Moodle multi-tenant globals
client_max_body_size 256M;
client_body_buffer_size 128k;
client_header_buffer_size 4k;
large_client_header_buffers 4 16k;
keepalive_timeout 65;
keepalive_requests 100;
server_tokens off;

# Gzip
gzip on;
gzip_vary on;
gzip_proxied any;
gzip_comp_level 5;
gzip_min_length 256;
gzip_types
    application/atom+xml application/javascript application/json
    application/ld+json application/manifest+json application/rss+xml
    application/vnd.geo+json application/vnd.ms-fontobject
    application/x-font-ttf application/x-web-app-manifest+json
    application/xhtml+xml application/xml font/opentype image/bmp
    image/svg+xml image/x-icon text/cache-manifest text/css
    text/plain text/vcard text/vnd.rim.location.xloc
    text/vtt text/x-component text/x-cross-domain-policy;

# Login bruteforce rate-limit (used per-site via limit_req)
limit_req_zone $binary_remote_addr zone=moodle_login:10m rate=5r/m;
EOF

    # Remove default site
    rm -f /etc/nginx/sites-enabled/default

    nginx -t
    systemctl enable --now nginx
    systemctl reload nginx
    ok "Nginx installed with global Moodle tuning"
}

#==============================================================================
# 5. Redis — shared sessions + MUC cache for all Moodle sites
#==============================================================================
step_redis() {
    section "STEP 5/9 — Redis (shared sessions + MUC cache)"

    apt-get install -y -qq redis-server

    # ---- Tuning ----
    local redis_max=$(( TOTAL_RAM_GB * 64 ))   # ~6% of RAM, ~1GB on 16GB box
    sed -i \
        -e 's|^bind .*|bind 127.0.0.1 -::1|' \
        -e 's|^# *maxmemory .*|maxmemory '${redis_max}'mb|' \
        -e 's|^maxmemory .*|maxmemory '${redis_max}'mb|' \
        -e 's|^# *maxmemory-policy .*|maxmemory-policy allkeys-lru|' \
        -e 's|^maxmemory-policy .*|maxmemory-policy allkeys-lru|' \
        -e 's|^appendonly .*|appendonly no|' \
        -e 's|^save .*||' \
        /etc/redis/redis.conf

    # Ensure tuning is applied even if directives weren't present
    grep -q "^maxmemory " /etc/redis/redis.conf || echo "maxmemory ${redis_max}mb" >> /etc/redis/redis.conf
    grep -q "^maxmemory-policy " /etc/redis/redis.conf || echo "maxmemory-policy allkeys-lru" >> /etc/redis/redis.conf

    systemctl enable --now redis-server
    systemctl restart redis-server
    sleep 1
    redis-cli ping | grep -q PONG
    ok "Redis running on 127.0.0.1:6379 (maxmemory=${redis_max}mb, allkeys-lru)"
}

#==============================================================================
# 6. Certbot
#==============================================================================
step_certbot() {
    section "STEP 6/9 — Certbot (Let's Encrypt)"

    apt-get install -y -qq certbot python3-certbot-nginx
    systemctl enable --now certbot.timer
    ok "Certbot installed; auto-renewal timer enabled"
}

#==============================================================================
# 7. Firewall + fail2ban
#==============================================================================
step_security() {
    section "STEP 7/9 — UFW firewall + fail2ban"

    # UFW
    ufw --force reset >/dev/null
    ufw default deny incoming
    ufw default allow outgoing
    ufw allow OpenSSH
    ufw allow 'Nginx Full'
    ufw --force enable

    # fail2ban — ssh + nginx login bruteforce
    cat > /etc/fail2ban/jail.local <<'EOF'
[DEFAULT]
bantime = 1h
findtime = 10m
maxretry = 5

[sshd]
enabled = true

[nginx-http-auth]
enabled = true

[nginx-limit-req]
enabled = true
filter = nginx-limit-req
logpath = /var/log/nginx/*error.log
maxretry = 10
EOF

    systemctl enable --now fail2ban
    systemctl restart fail2ban
    ok "UFW enabled (22, 80, 443); fail2ban active"
}

#==============================================================================
# 8. Cron daemon (per-site cron entries are added by provision-site.sh)
#==============================================================================
step_cron() {
    section "STEP 8/9 — Cron daemon"
    systemctl enable --now cron
    # Per-site cron entries go in /etc/cron.d/moodle-<slug>
    ok "Cron daemon ready (per-site jobs added on provisioning)"
}

#==============================================================================
# 9. State directory + finalise
#==============================================================================
step_state() {
    section "STEP 9/9 — Multi-tenant state directory"
    mkdir -p "$STATE_DIR/sites"
    chmod 700 "$STATE_DIR"

    cat > "$STATE_DIR/global.conf" <<EOF
# Moodle multi-tenant global config — generated $(date)
BASE_DOMAIN="$BASE_DOMAIN"
LETSENCRYPT_EMAIL="$LETSENCRYPT_EMAIL"
PHP_VERSION="$PHP_VERSION"
PG_VERSION="$PG_VERSION"
TOTAL_RAM_GB="$TOTAL_RAM_GB"
EOF
    chmod 600 "$STATE_DIR/global.conf"

    date '+%Y-%m-%d %H:%M:%S' > "$STATE_DIR/.bootstrap_complete"
    ok "State directory ready at $STATE_DIR"
}

#==============================================================================
# Final summary
#==============================================================================
print_summary() {
    local server_ip
    server_ip=$(curl -fsSL https://api.ipify.org 2>/dev/null || hostname -I | awk '{print $1}')
    cat <<EOF

================================================================================
$(echo -e "${GREEN}MOODLE 5.2 MULTI-TENANT BOOTSTRAP COMPLETE${NC}")
================================================================================

Server IP            : $server_ip
Base domain          : $BASE_DOMAIN
PHP                  : $PHP_VERSION (Moodle 5.2 supported)
PostgreSQL           : $PG_VERSION
Redis                : 127.0.0.1:6379
State directory      : $STATE_DIR
Bootstrap log        : $LOG_FILE

REQUIRED NEXT STEP — DNS:
   Add an A-record at your DNS provider:
       *.${BASE_DOMAIN}   →   $server_ip
   This lets every <slug>.${BASE_DOMAIN} subdomain resolve automatically.
   Verify with: dig +short test123.${BASE_DOMAIN}

PROVISIONING A SITE:
   sudo ./provision-site.sh <slug> [admin-email]

   Example:
       sudo ./provision-site.sh demo
       → creates https://demo.${BASE_DOMAIN}

   Slug rules: lowercase, alphanumeric + hyphens, 3-20 chars.

================================================================================
EOF
}

#==============================================================================
# Main
#==============================================================================
main() {
    require_root
    already_done

    info "Starting Moodle 5.2 multi-tenant bootstrap"
    info "Logging to $LOG_FILE"

    step_system
    step_php
    step_postgres
    step_nginx
    step_redis
    step_certbot
    step_security
    step_cron
    step_state

    print_summary
}

main "$@"
