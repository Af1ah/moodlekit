#!/bin/bash
#==============================================================================
# Moodle 5.2 multi-site provisioner for Nginx + PHP-FPM + PostgreSQL
#
# Assumptions:
# - One Moodle codebase per site under /var/www/moodle/<slug>
# - One dataroot per site under /var/moodledata/<slug>
# - One PostgreSQL database and login role per site
# - One PHP-FPM pool and one Nginx vhost per site
#
# Primary references used while shaping this script:
# - Moodle Nginx docs: https://docs.moodle.org/502/en/Nginx
# - Moodle Router docs: https://docs.moodle.org/502/en/Configuring_the_Router
# - Moodle install docs: https://docs.moodle.org/502/en/Installing_Moodle
# - Moodle config-dist.php comments for X-Sendfile/router behavior
# - PostgreSQL CREATE ROLE docs: https://www.postgresql.org/docs/18/sql-createrole.html
# - PostgreSQL schema security docs: https://www.postgresql.org/docs/18/ddl-schemas.html
#==============================================================================

set -euo pipefail

WEB_USER="${WEB_USER:-www-data}"
WEB_GROUP="${WEB_GROUP:-www-data}"
PHP_VERSION="${PHP_VERSION:-8.3}"
BASE_DOMAIN="${BASE_DOMAIN:-ariise.cloud}"
LETSENCRYPT_EMAIL="${LETSENCRYPT_EMAIL:-admin@${BASE_DOMAIN}}"
MOODLE_BRANCH="${MOODLE_BRANCH:-MOODLE_502_STABLE}"
MOODLE_GIT_URL="${MOODLE_GIT_URL:-https://github.com/moodle/moodle.git}"
STATE_DIR="${STATE_DIR:-/etc/moodle-multi}"
SITE_ROOT="${SITE_ROOT:-/var/www/moodle}"
DATA_ROOT="${DATA_ROOT:-/var/moodledata}"
PHP_BIN="${PHP_BIN:-/usr/bin/php${PHP_VERSION}}"
PGHOST="${PGHOST:-localhost}"
PGPORT="${PGPORT:-5432}"
DB_PREFIX="${DB_PREFIX:-mdl_}"
DIRECTORY_PERMISSIONS="${DIRECTORY_PERMISSIONS:-02770}"
ENABLE_REDIS_SESSIONS="${ENABLE_REDIS_SESSIONS:-0}"
ALLOW_PLUGIN_UI_INSTALL="${ALLOW_PLUGIN_UI_INSTALL:-0}"
ENABLE_CERTBOT="${ENABLE_CERTBOT:-1}"
SKIP_DNS_CHECK="${SKIP_DNS_CHECK:-0}"

LOG_FILE="/var/log/moodle-provision-$(date +%Y%m%d_%H%M%S).log"

RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
BLUE='\033[0;34m'
NC='\033[0m'

log()     { echo -e "$(date '+%F %T') [$1] $2" | tee -a "$LOG_FILE"; }
info()    { echo -e "${BLUE}[INFO]${NC}  $1"; log INFO "$1"; }
ok()      { echo -e "${GREEN}[ OK ]${NC}  $1"; log OK "$1"; }
warn()    { echo -e "${YELLOW}[WARN]${NC}  $1"; log WARN "$1"; }
err()     { echo -e "${RED}[FAIL]${NC}  $1"; log FAIL "$1"; }
section() { echo; echo -e "${BLUE}==>${NC} $1"; log STEP "$1"; }

trap 'err "Provision failed at line $LINENO (exit $?). See $LOG_FILE"' ERR

usage() {
    cat <<EOF
Usage: sudo $0 <slug> [admin-email]

Environment overrides:
  BASE_DOMAIN              Default: ${BASE_DOMAIN}
  LETSENCRYPT_EMAIL        Default: ${LETSENCRYPT_EMAIL}
  PHP_VERSION              Default: ${PHP_VERSION}
  MOODLE_BRANCH            Default: ${MOODLE_BRANCH}
  ENABLE_CERTBOT           Default: ${ENABLE_CERTBOT} (1 or 0)
  ENABLE_REDIS_SESSIONS    Default: ${ENABLE_REDIS_SESSIONS} (1 or 0)
  ALLOW_PLUGIN_UI_INSTALL  Default: ${ALLOW_PLUGIN_UI_INSTALL} (1 or 0)
  SKIP_DNS_CHECK           Default: ${SKIP_DNS_CHECK} (1 or 0)

Example:
  sudo BASE_DOMAIN=ariise.cloud $0 kmoocs admin@ariise.cloud
EOF
}

[[ $EUID -eq 0 ]] || { err "Must run as root"; exit 1; }

if [[ $# -lt 1 ]]; then
    usage
    exit 1
fi

SLUG="$1"
ADMIN_EMAIL="${2:-$LETSENCRYPT_EMAIL}"

if [[ ! "$SLUG" =~ ^[a-z][a-z0-9-]{2,19}$ ]]; then
    err "Invalid slug '$SLUG': use 3-20 chars, lowercase a-z, 0-9, hyphen; must start with a letter"
    exit 1
fi

DOMAIN="${SLUG}.${BASE_DOMAIN}"
SITE_DIR="${SITE_ROOT}/${SLUG}"
DATA_DIR="${DATA_ROOT}/${SLUG}"
DB_NAME="moodle_${SLUG}"
DB_USER="moodle_${SLUG}"
FPM_POOL="moodle_${SLUG}"
FPM_SOCK="/run/php/php${PHP_VERSION}-fpm-${SLUG}.sock"
FPM_CONF="/etc/php/${PHP_VERSION}/fpm/pool.d/${FPM_POOL}.conf"
NGINX_CONF="/etc/nginx/sites-available/moodle-${SLUG}"
NGINX_LINK="/etc/nginx/sites-enabled/moodle-${SLUG}"
CRON_FILE="/etc/cron.d/moodle-${SLUG}"
SITE_CONF="${STATE_DIR}/sites/${SLUG}.conf"
SITE_URL="https://${DOMAIN}"
INSTALL_WWWROOT="$SITE_URL"

if [[ "$BASE_DOMAIN" == "localhost" ]]; then
    ENABLE_CERTBOT=0
    SKIP_DNS_CHECK=1
    SITE_URL="http://${DOMAIN}"
    INSTALL_WWWROOT="$SITE_URL"
    if [[ "$ADMIN_EMAIL" == "$LETSENCRYPT_EMAIL" ]]; then
        ADMIN_EMAIL="admin@example.com"
    fi
fi

gen_password() {
    openssl rand -base64 24 | tr -d '/+=' | head -c 24
}

DB_PASS="$(gen_password)"
ADMIN_USER="admin"
ADMIN_PASS="$(gen_password)"

ensure_dependencies() {
    section "STEP 1/12 — Dependency checks"

    local required=(
        git
        nginx
        psql
        openssl
        curl
        getent
        awk
        sed
        tee
        systemctl
        "$PHP_BIN"
    )

    if [[ "$ENABLE_CERTBOT" == "1" ]]; then
        required+=(certbot)
    fi

    local missing=0
    for cmd in "${required[@]}"; do
        if ! command -v "$cmd" >/dev/null 2>&1; then
            err "Missing required command: $cmd"
            missing=1
        fi
    done

    if [[ "$ALLOW_PLUGIN_UI_INSTALL" == "1" ]] && ! command -v setfacl >/dev/null 2>&1; then
        err "ALLOW_PLUGIN_UI_INSTALL=1 requires setfacl"
        missing=1
    fi

    if [[ "$ENABLE_REDIS_SESSIONS" == "1" ]] && ! command -v redis-cli >/dev/null 2>&1; then
        err "ENABLE_REDIS_SESSIONS=1 requires redis-cli"
        missing=1
    fi

    if [[ "$missing" -ne 0 ]]; then
        exit 1
    fi

    mkdir -p "${STATE_DIR}/sites"
    chmod 0750 "${STATE_DIR}" "${STATE_DIR}/sites"
    ok "Dependencies and state directory verified"
}

check_existing_site() {
    section "STEP 2/12 — Existing site checks"

    if [[ -e "$SITE_CONF" || -d "$SITE_DIR" || -d "$DATA_DIR" || -e "$NGINX_CONF" || -e "$FPM_CONF" ]]; then
        err "Site '${SLUG}' appears to already exist"
        err "Check: $SITE_CONF, $SITE_DIR, $DATA_DIR, $NGINX_CONF, $FPM_CONF"
        exit 1
    fi

    ok "Slug is available"
}

dns_check() {
    section "STEP 3/12 — DNS check"

    if [[ "$SKIP_DNS_CHECK" == "1" ]]; then
        warn "Skipping DNS check"
        return
    fi

    local server_ip resolved_ip
    server_ip="$(curl -4fsSL https://api.ipify.org 2>/dev/null || hostname -I | awk '{print $1}')"
    resolved_ip="$(getent ahostsv4 "$DOMAIN" | awk '/STREAM/ {print $1; exit}')"

    if [[ -z "$resolved_ip" ]]; then
        err "DNS for $DOMAIN does not resolve yet"
        exit 1
    fi

    if [[ "$resolved_ip" != "$server_ip" ]]; then
        warn "DNS for $DOMAIN resolves to $resolved_ip but this server IP appears to be $server_ip"
        warn "Continuing, but certificate issuance may fail until DNS propagates"
    else
        ok "DNS: $DOMAIN -> $resolved_ip"
    fi
}

create_database() {
    section "STEP 4/12 — PostgreSQL database and role"

    sudo -u postgres psql <<SQL
CREATE ROLE ${DB_USER}
    LOGIN
    NOSUPERUSER
    NOCREATEDB
    NOCREATEROLE
    NOINHERIT
    NOREPLICATION
    PASSWORD '${DB_PASS}';

CREATE DATABASE ${DB_NAME}
    WITH OWNER = ${DB_USER}
    ENCODING = 'UTF8';

REVOKE ALL ON DATABASE ${DB_NAME} FROM PUBLIC;
GRANT CONNECT, TEMP ON DATABASE ${DB_NAME} TO ${DB_USER};
\connect ${DB_NAME}
REVOKE CREATE ON SCHEMA public FROM PUBLIC;
GRANT USAGE, CREATE ON SCHEMA public TO ${DB_USER};
ALTER SCHEMA public OWNER TO ${DB_USER};
SQL

    ok "Database ${DB_NAME} and least-privilege login role ${DB_USER} created"
}

clone_moodle() {
    section "STEP 5/12 — Clone Moodle"

    mkdir -p "$SITE_ROOT"
    git clone --depth 1 --branch "$MOODLE_BRANCH" "$MOODLE_GIT_URL" "$SITE_DIR"
    ok "Cloned ${MOODLE_BRANCH} into $SITE_DIR"

    if [[ -f "${SITE_DIR}/composer.json" ]]; then
        if ! command -v composer >/dev/null 2>&1; then
            warn "Composer not found; skipping composer install"
        else
            info "Running composer install"
            chown -R "${WEB_USER}:${WEB_GROUP}" "$SITE_DIR"
            (
                cd "$SITE_DIR"
                sudo -u "$WEB_USER" HOME=/tmp composer install \
                    --no-dev \
                    --classmap-authoritative \
                    --no-interaction \
                    --no-progress
            ) 2>&1 | tee -a "$LOG_FILE"
            chown -R root:"$WEB_GROUP" "$SITE_DIR"
            ok "Composer dependencies installed"
        fi
    fi
}

set_permissions() {
    section "STEP 6/12 — Filesystem layout and permissions"

    mkdir -p "$DATA_DIR"

    chown -R "${WEB_USER}:${WEB_GROUP}" "$DATA_DIR"
    chmod 02770 "$DATA_DIR"

    chown -R root:"$WEB_GROUP" "$SITE_DIR"
    find "$SITE_DIR" -type d -exec chmod 0755 {} \;
    find "$SITE_DIR" -type f -exec chmod 0644 {} \;

    if [[ "$ALLOW_PLUGIN_UI_INSTALL" == "1" ]]; then
        local writable_components=(
            public/mod
            public/blocks
            public/local
            public/theme
            public/auth
            public/filter
            public/admin/tool
            public/question/type
            public/course/format
            public/repository
            public/enrol
            public/customfield/field
            public/availability/condition
            public/message/output
            public/portfolio
            public/lib/editor/tiny/plugins
        )

        for rel in "${writable_components[@]}"; do
            if [[ -d "${SITE_DIR}/${rel}" ]]; then
                setfacl -R -m "u:${WEB_USER}:rwX" "${SITE_DIR}/${rel}"
                setfacl -R -d -m "u:${WEB_USER}:rwX" "${SITE_DIR}/${rel}"
            fi
        done

        ok "Plugin UI install ACLs enabled"
    else
        info "Plugin UI install ACLs skipped; deploy plugin code as root/git instead"
    fi

    ok "Code and dataroot permissions applied"
}

create_fpm_pool() {
    section "STEP 7/12 — PHP-FPM pool"

    mkdir -p /var/log/php-fpm
    chown "${WEB_USER}:${WEB_GROUP}" /var/log/php-fpm

    cat > "$FPM_CONF" <<EOF
[${FPM_POOL}]
user = ${WEB_USER}
group = ${WEB_GROUP}
listen = ${FPM_SOCK}
listen.owner = ${WEB_USER}
listen.group = ${WEB_GROUP}
listen.mode = 0660

pm = dynamic
pm.max_children = 12
pm.start_servers = 3
pm.min_spare_servers = 2
pm.max_spare_servers = 5
pm.max_requests = 500

request_terminate_timeout = 300s
request_slowlog_timeout = 10s
slowlog = /var/log/php-fpm/${SLUG}-slow.log

php_admin_value[memory_limit] = 384M
php_admin_value[max_execution_time] = 300
php_admin_value[max_input_vars] = 5000
php_admin_value[upload_max_filesize] = 256M
php_admin_value[post_max_size] = 256M
php_admin_value[error_log] = /var/log/php-fpm/${SLUG}-error.log
php_admin_flag[log_errors] = on
security.limit_extensions = .php
EOF

    systemctl reload "php${PHP_VERSION}-fpm"
    local waited=0
    while [[ ! -S "$FPM_SOCK" && "$waited" -lt 15 ]]; do
        sleep 1
        waited=$((waited + 1))
    done

    [[ -S "$FPM_SOCK" ]] || { err "PHP-FPM socket not created after ${waited}s: $FPM_SOCK"; exit 1; }
    ok "Dedicated PHP-FPM pool active"
}

write_http_nginx_config() {
    cat > "$NGINX_CONF" <<EOF
server {
    listen 80;
    listen [::]:80;
    server_name ${DOMAIN};

    root ${SITE_DIR}/public;
    index index.php;
    client_max_body_size 256M;

    access_log /var/log/nginx/${SLUG}_access.log;
    error_log /var/log/nginx/${SLUG}_error.log;

    location / {
        try_files \$uri \$uri/ /r.php\$is_args\$args;
    }

    location ~ \.php(/|\$) {
        fastcgi_split_path_info ^(.+\.php)(/.*)\$;
        set \$path_info \$fastcgi_path_info;
        try_files \$fastcgi_script_name \$fastcgi_script_name/ /r.php\$is_args\$args;

        fastcgi_pass unix:${FPM_SOCK};
        include fastcgi_params;
        fastcgi_param PATH_INFO \$path_info;
        fastcgi_param SCRIPT_FILENAME \$realpath_root\$fastcgi_script_name;
        fastcgi_param DOCUMENT_ROOT \$realpath_root;
        fastcgi_read_timeout 300;
    }

    location ~ /\.(?!well-known).* {
        return 404;
    }

    location ~ (/vendor/|/node_modules/|composer\.json|/readme|/README|readme\.txt|/upgrade\.txt|/UPGRADING\.md|db/install\.xml|/fixtures/|/behat/|phpunit\.xml|\.lock|environment\.xml) {
        deny all;
        return 404;
    }

    location ^~ /.well-known/acme-challenge/ {
        default_type "text/plain";
        allow all;
    }

    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;
}
EOF
}

write_https_nginx_config() {
    cat > "$NGINX_CONF" <<EOF
server {
    listen 80;
    listen [::]:80;
    server_name ${DOMAIN};
    return 301 https://\$host\$request_uri;
}

server {
    listen 443 ssl http2;
    listen [::]:443 ssl http2;
    server_name ${DOMAIN};

    root ${SITE_DIR}/public;
    index index.php;
    client_max_body_size 256M;

    access_log /var/log/nginx/${SLUG}_access.log;
    error_log /var/log/nginx/${SLUG}_error.log;

    ssl_certificate /etc/letsencrypt/live/${DOMAIN}/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/${DOMAIN}/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    location / {
        try_files \$uri \$uri/ /r.php\$is_args\$args;
    }

    location ~ \.php(/|\$) {
        fastcgi_split_path_info ^(.+\.php)(/.*)\$;
        set \$path_info \$fastcgi_path_info;
        try_files \$fastcgi_script_name \$fastcgi_script_name/ /r.php\$is_args\$args;

        fastcgi_pass unix:${FPM_SOCK};
        include fastcgi_params;
        fastcgi_param HTTPS on;
        fastcgi_param PATH_INFO \$path_info;
        fastcgi_param SCRIPT_FILENAME \$realpath_root\$fastcgi_script_name;
        fastcgi_param DOCUMENT_ROOT \$realpath_root;
        fastcgi_read_timeout 300;
    }

    location ~ /\.(?!well-known).* {
        return 404;
    }

    location ~ (/vendor/|/node_modules/|composer\.json|/readme|/README|readme\.txt|/upgrade\.txt|/UPGRADING\.md|db/install\.xml|/fixtures/|/behat/|phpunit\.xml|\.lock|environment\.xml) {
        deny all;
        return 404;
    }

    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;
}
EOF
}

configure_nginx_http() {
    section "STEP 8/12 — HTTP Nginx vhost"

    write_http_nginx_config
    ln -s "$NGINX_CONF" "$NGINX_LINK"
    nginx -t
    systemctl reload nginx
    ok "HTTP vhost active"
}

configure_tls() {
    section "STEP 9/12 — TLS certificate and HTTPS vhost"

    if [[ "$ENABLE_CERTBOT" != "1" ]]; then
        warn "ENABLE_CERTBOT=0, keeping HTTP-only setup"
        INSTALL_WWWROOT="http://${DOMAIN}"
        return
    fi

    certbot certonly \
        --webroot \
        --non-interactive \
        --agree-tos \
        --email "$ADMIN_EMAIL" \
        -w "${SITE_DIR}/public" \
        -d "$DOMAIN" \
        2>&1 | tee -a "$LOG_FILE"

    write_https_nginx_config
    nginx -t
    systemctl reload nginx
    ok "TLS enabled for $DOMAIN"
}

install_moodle() {
    section "STEP 10/12 — Moodle CLI install"

    chown "${WEB_USER}:${WEB_GROUP}" "$SITE_DIR"

    sudo -u "$WEB_USER" "$PHP_BIN" "${SITE_DIR}/admin/cli/install.php" \
        --non-interactive \
        --agree-license \
        --lang=en \
        --wwwroot="${INSTALL_WWWROOT}" \
        --dataroot="${DATA_DIR}" \
        --dbtype=pgsql \
        --dbhost="${PGHOST}" \
        --dbport="${PGPORT}" \
        --dbname="${DB_NAME}" \
        --dbuser="${DB_USER}" \
        --dbpass="${DB_PASS}" \
        --prefix="${DB_PREFIX}" \
        --fullname="${SLUG^} Moodle" \
        --shortname="${SLUG^}" \
        --summary="Moodle 5.2 instance for ${SLUG}" \
        --adminuser="${ADMIN_USER}" \
        --adminpass="${ADMIN_PASS}" \
        --adminemail="${ADMIN_EMAIL}" \
        2>&1 | tee -a "$LOG_FILE"

    chown root:"$WEB_GROUP" "$SITE_DIR"

    [[ -f "${SITE_DIR}/config.php" ]] || { err "config.php was not created"; exit 1; }
    ok "Moodle installed successfully"
}

patch_config() {
    section "STEP 11/12 — Final config hardening"

    local cfg="${SITE_DIR}/config.php"
    local tmp
    tmp="$(mktemp)"

    awk -v add="$(cat <<EOF

// === moodle-multi additions ===
\$CFG->directorypermissions = ${DIRECTORY_PERMISSIONS};
\$CFG->routerconfigured = true;
EOF
)" '
        /^require_once/ && !done { print add; done=1 }
        { print }
    ' "$cfg" > "$tmp"

    mv "$tmp" "$cfg"

    if [[ "$ENABLE_REDIS_SESSIONS" == "1" ]]; then
        if redis-cli ping >/dev/null 2>&1; then
            tmp="$(mktemp)"
            awk -v add="$(cat <<EOF
\$CFG->session_handler_class = '\\\\core\\\\session\\\\redis';
\$CFG->session_redis_host = '127.0.0.1';
\$CFG->session_redis_port = 6379;
\$CFG->session_redis_database = 0;
\$CFG->session_redis_prefix = 'mdl_${SLUG}_sess_';
\$CFG->session_redis_acquire_lock_timeout = 120;
\$CFG->session_redis_lock_expire = 7200;
EOF
)" '
                /\/\/ === moodle-multi additions ===/ && !done { print; print add; done=1; next }
                { print }
            ' "$cfg" > "$tmp"
            mv "$tmp" "$cfg"
            ok "Redis session configuration added"
        else
            warn "Redis ping failed; leaving session configuration unchanged"
        fi
    fi

    chown root:"$WEB_GROUP" "$cfg"
    chmod 0640 "$cfg"

    sudo -u "$WEB_USER" "$PHP_BIN" "${SITE_DIR}/admin/cli/purge_caches.php" \
        2>&1 | tee -a "$LOG_FILE"

    ok "config.php hardened and caches purged"
}

configure_cron() {
    section "STEP 12/12 — Cron"

    local existing offset lock_file cron_php
    existing="$(find "${STATE_DIR}/sites" -maxdepth 1 -name '*.conf' -type f 2>/dev/null | wc -l)"
    offset="$(( (existing * 12) % 60 ))"

    lock_file="/tmp/moodle-${SITE_SLUG}.lock"
    cron_php="${SITE_DIR}/admin/cli/cron.php"

    if [ ! -f "$cron_php" ]; then
        die "Moodle cron.php not found: $cron_php"
    fi

    cat > "$CRON_FILE" <<EOF
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
* * * * * ${WEB_USER} sleep ${offset}; /usr/bin/flock -n ${lock_file} ${PHP_BIN} ${cron_php} >/dev/null 2>&1
EOF

    chown root:root "$CRON_FILE"
    chmod 0644 "$CRON_FILE"

    ok "Cron installed with ${offset}s offset and lock: ${lock_file}"
}

configure_moodle_task_processing() {
    section "Configure Moodle task processing"

    local cfg_php="${SITE_DIR}/admin/cli/cfg.php"

    if [ ! -f "$cfg_php" ]; then
        warn "cfg.php not found, skipping cron_keepalive setting"
        return 0
    fi

    sudo -u "$WEB_USER" "$PHP_BIN" "$cfg_php" --name=cron_keepalive --set=0

    ok "Moodle cron_keepalive set to 0"
}

save_site_conf() {
    cat > "$SITE_CONF" <<EOF
SLUG="${SLUG}"
DOMAIN="${DOMAIN}"
URL="${INSTALL_WWWROOT}"
SITE_DIR="${SITE_DIR}"
DATA_DIR="${DATA_DIR}"
DB_NAME="${DB_NAME}"
DB_USER="${DB_USER}"
DB_PASS="${DB_PASS}"
ADMIN_USER="${ADMIN_USER}"
ADMIN_PASS="${ADMIN_PASS}"
ADMIN_EMAIL="${ADMIN_EMAIL}"
FPM_POOL="${FPM_POOL}"
FPM_SOCK="${FPM_SOCK}"
NGINX_CONF="${NGINX_CONF}"
CRON_FILE="${CRON_FILE}"
LOG_FILE="${LOG_FILE}"
EOF

    chmod 0600 "$SITE_CONF"
    ok "Provision details saved to $SITE_CONF"
}

print_summary() {
    cat <<EOF

================================================================================
Provisioned Moodle site: ${SLUG}
================================================================================
URL:           ${INSTALL_WWWROOT}
Admin user:    ${ADMIN_USER}
Admin pass:    ${ADMIN_PASS}
Admin email:   ${ADMIN_EMAIL}
Code dir:      ${SITE_DIR}
Data dir:      ${DATA_DIR}
Database:      ${DB_NAME}
DB user:       ${DB_USER}
PHP-FPM pool:  ${FPM_POOL}
Nginx config:  ${NGINX_CONF}
Cron file:     ${CRON_FILE}
State file:    ${SITE_CONF}
Provision log: ${LOG_FILE}

Notes:
- Router support is enabled via Nginx try_files and \$CFG->routerconfigured = true.
- X-Accel-Redirect is intentionally not enabled by default.
- Plugin UI installation ACLs are disabled unless ALLOW_PLUGIN_UI_INSTALL=1.
================================================================================
EOF
}

main() {
    info "Provisioning Moodle site ${SLUG} at ${DOMAIN}"
    info "Provision log: ${LOG_FILE}"

    ensure_dependencies
    check_existing_site
    dns_check
    create_database
    clone_moodle
    set_permissions
    create_fpm_pool
    configure_nginx_http
    configure_tls
    install_moodle
    patch_config
    configure_cron
    save_site_conf
    print_summary
}

main "$@"
